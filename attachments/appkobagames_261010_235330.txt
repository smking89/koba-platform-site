5. PRISMA DATABASE MODEL SCHEMAS
Provide the explicit Prisma database schema models required to support this app store engine, tracking staff approval workflows, financial splits, and verification badges:

prisma
enum Role {
  PLAYER
  BUSINESS
  STAFF
}

enum AppStatus {
  PENDING_REVIEW
  APPROVED
  REJECTED
}

model User {
  id                 String       @id @default(uuid())
  kobaId             String       @unique // The unique system-wide fingerprint
  email              String       @unique
  role               Role         @default(PLAYER)
  isVerifiedDeveloper Boolean      @default(false) // Controls 2.5% (true) vs 3.0% (false) fee tier
  purchases          Purchase[]
  apps               AppListing[]
  createdAt          DateTime     @default(now())
}

model AppListing {
  id            String          @id @default(uuid())
  name          String
  developerId   String
  developer     User            @relation(fields: [developerId], references: [id])
  type          String          // "BOT", "PLUGIN", "ADAPTER"
  game          String          // "RUST", "DAYZ", "ARK", "MINECRAFT"
  rarityTier    String          // "COMMON" through "RELIC"
  billingModel  String          // "BUY_NOW" or "SUBSCRIPTION"
  basePrice     Int             // Stored in cents (\$10.00 = 1000)
  metaUrl       String          // Content or storage path
  status        AppStatus       @default(PENDING_REVIEW) // Strictly requires manual STAFF approval
  verifiedBy    String?         // Stores the KOBAid of the staff member who approved it
  purchases     Purchase[]
  tokens        DownloadToken[]
  createdAt     DateTime        @default(now())
}

model Purchase {
  id              String     @id @default(uuid())
  userId          String
  user            User       @relation(fields: [userId], references: [id])
  appId           String
  app             AppListing @relation(fields: [appId], references: [id])
  licenseKey      String     @unique
  billingModel    String     
  amountPaidTotal Int        
  platformFeeCharged Int     
  developerPayout Int        
  subscriptionEnd DateTime?  
  isActive        Boolean    @default(true)
  createdAt       DateTime   @default(now())
}

model DownloadToken {
  id          String     @id @default(uuid())
  tokenString String     @unique
  appId       String
  app         AppListing @relation(fields: [appId], references: [id])
  kobaId      String     
  isUsed      Boolean    @default(false)
  expiresAt   DateTime
  createdAt   DateTime   @default(now())
}
